/**

Neon Teleporter VTO Engine
Handles Camera, Upload, Model modes with Canvas Compositing.
*/
class NeonVTO {
constructor() {
this.data = this.parseData();
this.els = {
modal: document.getElementById('vto-modal'),
closeBtn: document.querySelector('.vto-close-btn'),
startScreen: document.getElementById('vto-start-screen'),
canvasScreen: document.getElementById('vto-canvas-screen'),
tiles: document.querySelectorAll('.vto-tile'),
fileInput: document.getElementById('vto-file-input'),
canvas: document.getElementById('vto-canvas'),
video: document.getElementById('vto-video'),
loader: document.getElementById('vto-loader'),
status: document.getElementById('vto-status-text'),
scaleSlider: document.getElementById('vto-scale-slider'),
ySlider: document.getElementById('vto-y-slider'),
resetBtn: document.getElementById('vto-reset-btn'),
snapBtn: document.getElementById('vto-snap-btn'),
addToCartBtn: document.getElementById('vto-add-cart-btn'),
modelSelect: document.getElementById('vto-model-select'),
modelGroup: document.getElementById('vto-model-select-group')
};

    this.ctx = this.els.canvas.getContext('2d');
    this.state = {
        mode: null, // 'camera', 'upload', 'model'
        active: false,
        variantId: document.querySelector('input[name="id"]')?.value || null,
        imgOverlay: null,
        imgBackground: null,
        videoStream: null,
        overlayTransform: { x: 0, y: 0, scale: 1.0, rotation: 0 },
        isDragging: false,
        lastMouse: { x: 0, y: 0 }
    };

    this.init();
}

parseData() {
    const el = document.getElementById('vto-product-data');
    try {
        return el ? JSON.parse(el.textContent) : {};
    } catch (e) {
        console.error('VTO Data Parse Error', e);
        return {};
    }
}

init() {
    this.setupEventListeners();
    this.populateModels();
    
    // Listen for global variant changes from the main product page script
    const variantInput = document.querySelector('input[name="id"]');
    if(variantInput) {
        // Use MutationObserver for hidden input changes if custom events aren't available
        const observer = new MutationObserver(() => {
            this.handleVariantChange(variantInput.value);
        });
        observer.observe(variantInput, { attributes: true, attributeFilter: ['value'] });
    }
    
    // Also listen to the radio buttons directly for faster feedback
    document.querySelectorAll('input[name="id"]').forEach(input => {
         input.addEventListener('change', (e) => this.handleVariantChange(e.target.value));
    });

    // Initial check if opened via CTA
    if(this.els.modal.classList.contains('is-visible')) {
        this.handleVariantChange(this.state.variantId);
    }
}

setupEventListeners() {
    // Modal Controls
    this.els.closeBtn.addEventListener('click', () => this.close());
    this.els.modal.addEventListener('click', (e) => {
        if (e.target === this.els.modal) this.close();
    });

    // Tiles
    this.els.tiles.forEach(tile => {
        tile.addEventListener('click', () => {
            const mode = tile.dataset.mode;
            this.startMode(mode);
        });
    });

    // File Input
    this.els.fileInput.addEventListener('change', (e) => this.handleFileUpload(e));

    // Canvas Interaction (Drag)
    this.els.canvas.addEventListener('mousedown', (e) => this.startDrag(e));
    this.els.canvas.addEventListener('mousemove', (e) => this.doDrag(e));
    this.els.canvas.addEventListener('mouseup', () => this.endDrag());
    this.els.canvas.addEventListener('mouseleave', () => this.endDrag());
    
    // Touch events
    this.els.canvas.addEventListener('touchstart', (e) => this.startDrag(e), {passive: false});
    this.els.canvas.addEventListener('touchmove', (e) => this.doDrag(e), {passive: false});
    this.els.canvas.addEventListener('touchend', () => this.endDrag());

    // Sliders
    this.els.scaleSlider.addEventListener('input', (e) => {
        this.state.overlayTransform.scale = parseFloat(e.target.value);
        this.draw();
    });
    this.els.ySlider.addEventListener('input', (e) => {
        this.state.overlayTransform.y = parseFloat(e.target.value);
        this.draw();
    });

    // Buttons
    this.els.resetBtn.addEventListener('click', () => this.resetTransform());
    this.els.snapBtn.addEventListener('click', () => this.saveSnapshot());
    this.els.addToCartBtn.addEventListener('click', () => this.addToCart());
    
    // Model Select
    this.els.modelSelect.addEventListener('change', (e) => {
        this.loadModelImage(e.target.value);
    });
}

populateModels() {
    const models = this.data.models || [];
    if (!models.length || !Array.isArray(models)) {
        // Fallback mock models if metafield is empty
        const mockModels = [
            { label: "Male Model 1", src: "https://cdn.shopify.com/s/files/1/0000/0001/files/model_m1.jpg?v=1" }, 
            { label: "Female Model 1", src: "https://cdn.shopify.com/s/files/1/0000/0001/files/model_f1.jpg?v=1" }
        ];
        // Only add if we have valid URLs, otherwise hide tile? 
        // For MVP we just assume metafields are populated or we show nothing.
    } else {
        this.els.modelSelect.innerHTML = models.map(m => 
            `<option value="${m.src}">${m.label}</option>`
        ).join('');
    }
}

close() {
    this.stopCamera();
    this.els.modal.classList.remove('is-visible');
    this.state.active = false;
    
    // Analytics
    this.trackEvent('vto_close');
}

async startMode(mode) {
    this.state.mode = mode;
    this.els.startScreen.style.display = 'none';
    this.els.canvasScreen.classList.add('is-active');
    this.state.active = true;

    this.trackEvent('vto_open', { mode: mode });

    if (mode === 'camera') {
        await this.startCamera();
    } else if (mode === 'upload') {
        this.els.fileInput.click();
    } else if (mode === 'model') {
        this.els.modelGroup.style.display = 'flex';
        if (this.els.modelSelect.options.length > 0) {
            this.loadModelImage(this.els.modelSelect.value);
        } else {
            alert("No models configured for this product.");
            this.close();
        }
    }

    // Load overlay for current variant
    this.loadOverlayForVariant(this.state.variantId);
}

async startCamera() {
    this.showLoader('Accessing Camera...');
    this.els.video.style.display = 'none'; // We render video to canvas manually
    
    try {
        const stream = await navigator.mediaDevices.getUserMedia({ 
            video: { facingMode: 'user', width: { ideal: 1280 }, height: { ideal: 720 } }, 
            audio: false 
        });
        this.state.videoStream = stream;
        this.els.video.srcObject = stream;
        await this.els.video.play();
        
        this.trackEvent('vto_camera_start');
        this.hideLoader();
        this.resizeCanvasToSource(this.els.video);
        this.loop();
    } catch (err) {
        console.error(err);
        alert('Camera access denied or unavailable. Please use "Upload Photo".');
        this.close(); // Or fallback
    }
}

stopCamera() {
    if (this.state.videoStream) {
        this.state.videoStream.getTracks().forEach(track => track.stop());
        this.state.videoStream = null;
    }
    cancelAnimationFrame(this.frameId);
}

handleFileUpload(e) {
    const file = e.target.files[0];
    if (!file) return;

    this.showLoader('Processing Image...');
    const reader = new FileReader();
    reader.onload = (event) => {
        const img = new Image();
        img.onload = () => {
            this.state.imgBackground = img;
            this.resizeCanvasToSource(img);
            this.hideLoader();
            this.trackEvent('vto_photo_upload');
            this.draw(); // Single draw for static image
        };
        img.src = event.target.result;
    };
    reader.readAsDataURL(file);
}

loadModelImage(src) {
    this.showLoader('Loading Model...');
    const img = new Image();
    img.crossOrigin = "Anonymous";
    img.onload = () => {
        this.state.imgBackground = img;
        this.resizeCanvasToSource(img);
        this.hideLoader();
        this.trackEvent('vto_model_select');
        this.draw();
    };
    img.src = src;
}

loadOverlayForVariant(variantId) {
    const variantData = this.data.variants[variantId];
    
    // Fallback: If no meta, use product main image? No, that looks bad overlaying a shirt on a shirt.
    // We need specific overlay assets.
    if (!variantData || !variantData.overlay) {
        console.warn('No VTO Overlay found for variant', variantId);
        // Optionally show a "Not supported" message on canvas
        this.state.imgOverlay = null;
        return;
    }

    const img = new Image();
    img.crossOrigin = "Anonymous";
    img.onload = () => {
        this.state.imgOverlay = img;
        // Center the overlay initially
        this.resetTransform(); 
        this.draw();
    };
    img.src = variantData.overlay;
}

handleVariantChange(newId) {
    if (this.state.variantId === newId) return;
    this.state.variantId = newId;
    if (this.state.active) {
        this.trackEvent('vto_variant_change', { variantId: newId });
        this.loadOverlayForVariant(newId);
    }
}

resizeCanvasToSource(source) {
    // Fit canvas to source aspect ratio within the container
    const container = this.els.canvas.parentElement;
    const contW = container.clientWidth;
    const contH = container.clientHeight;
    
    const srcW = source.videoWidth || source.naturalWidth;
    const srcH = source.videoHeight || source.naturalHeight;
    
    if (!srcW || !srcH) return;

    const ratio = Math.min(contW / srcW, contH / srcH);
    
    this.els.canvas.width = srcW;
    this.els.canvas.height = srcH;
    
    // CSS scaling for display
    this.els.canvas.style.width = `${srcW * ratio}px`;
    this.els.canvas.style.height = `${srcH * ratio}px`;
    
    // Reset transform based on new canvas size
    this.resetTransform();
}

resetTransform() {
    if (!this.els.canvas.width) return;
    // Logic to vaguely center garment on upper torso (30% down from top)
    this.state.overlayTransform = {
        x: this.els.canvas.width / 2,
        y: this.els.canvas.height * 0.4, 
        scale: 1.0,
        rotation: 0
    };
    this.els.scaleSlider.value = 1.0;
    this.els.ySlider.value = 0; // Relative to center
    this.draw();
}

// --- Interaction ---

getMousePos(e) {
    const rect = this.els.canvas.getBoundingClientRect();
    const clientX = e.touches ? e.touches[0].clientX : e.clientX;
    const clientY = e.touches ? e.touches[0].clientY : e.clientY;
    const scaleX = this.els.canvas.width / rect.width;
    const scaleY = this.els.canvas.height / rect.height;
    return {
        x: (clientX - rect.left) * scaleX,
        y: (clientY - rect.top) * scaleY
    };
}

startDrag(e) {
    if (!this.state.imgOverlay) return;
    e.preventDefault();
    this.state.isDragging = true;
    this.state.lastMouse = this.getMousePos(e);
}

doDrag(e) {
    if (!this.state.isDragging) return;
    e.preventDefault();
    const pos = this.getMousePos(e);
    const dx = pos.x - this.state.lastMouse.x;
    const dy = pos.y - this.state.lastMouse.y;
    
    this.state.overlayTransform.x += dx;
    this.state.overlayTransform.y += dy;
    this.state.lastMouse = pos;
    
    this.draw();
}

endDrag() {
    this.state.isDragging = false;
}

// --- Rendering ---

draw() {
    if (!this.ctx) return;
    const w = this.els.canvas.width;
    const h = this.els.canvas.height;
    this.ctx.clearRect(0, 0, w, h);

    // 1. Draw Background
    if (this.state.mode === 'camera' && this.state.videoStream) {
        this.ctx.save();
        this.ctx.translate(w, 0);
        this.ctx.scale(-1, 1); // Mirror camera
        this.ctx.drawImage(this.els.video, 0, 0, w, h);
        this.ctx.restore();
    } else if (this.state.imgBackground) {
        this.ctx.drawImage(this.state.imgBackground, 0, 0, w, h);
    }

    // 2. Draw Overlay
    if (this.state.imgOverlay) {
        const img = this.state.imgOverlay;
        const t = this.state.overlayTransform;
        
        // Apply Sliders
        const scale = t.scale * (w / 1200); // Normalize scale based on arbitrary base width
        const yOffset = parseInt(this.els.ySlider.value) || 0;

        const drawW = img.naturalWidth * scale;
        const drawH = img.naturalHeight * scale;

        this.ctx.save();
        this.ctx.translate(t.x, t.y + yOffset);
        this.ctx.rotate(t.rotation);
        this.ctx.globalAlpha = 0.95; // Slight transparency for blending
        // Draw centered
        this.ctx.drawImage(img, -drawW / 2, -drawH / 2, drawW, drawH);
        this.ctx.restore();
    }
}

loop() {
    if (this.state.mode === 'camera' && this.state.active) {
        this.draw();
        this.frameId = requestAnimationFrame(() => this.loop());
    }
}

// --- Utilities ---

showLoader(text) {
    this.els.loader.classList.remove('hidden');
    if (text) this.els.status.textContent = text;
}

hideLoader() {
    this.els.loader.classList.add('hidden');
}

saveSnapshot() {
    const link = document.createElement('a');
    link.download = `vto-snap-${Date.now()}.png`;
    link.href = this.els.canvas.toDataURL();
    link.click();
}

// --- Cart Integration ---

async addToCart() {
    const variantId = this.state.variantId;
    if (!variantId) return;

    this.els.addToCartBtn.textContent = 'Adding...';
    this.els.addToCartBtn.disabled = true;

    // Gather existing customization properties
    const properties = {};
    document.querySelectorAll('input[name^="properties["]').forEach(input => {
        if (!input.disabled && input.value) {
            const name = input.name.match(/\[(.*?)\]/)[1];
            properties[name] = input.value;
        }
    });

    // Add VTO properties
    properties['Virtual Try-On Used'] = 'Yes';
    properties['Virtual Try-On Mode'] = this.state.mode;
    properties['Virtual Try-On Timestamp'] = new Date().toISOString();
    if (this.state.mode === 'model') {
        properties['Virtual Try-On Model'] = this.els.modelSelect.options[this.els.modelSelect.selectedIndex].text;
    }

    const formData = {
        items: [{
            id: variantId,
            quantity: 1, // Default to 1
            properties: properties
        }]
    };

    try {
        const response = await fetch(window.Shopify.routes.root + 'cart/add.js', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(formData)
        });

        if (response.ok) {
            this.trackEvent('vto_add_cart');
            // Redirect to cart or trigger theme drawer
            window.location.href = window.Shopify.routes.root + 'cart';
        } else {
            throw new Error('Cart API Failed');
        }
    } catch (err) {
        console.error(err);
        alert('Failed to add to cart. Please try again.');
        this.els.addToCartBtn.textContent = 'Add Current Variant';
        this.els.addToCartBtn.disabled = false;
    }
}

trackEvent(name, data = {}) {
    // Analytics hook (GTM / Shopify Pixels)
    if (window.dataLayer) {
        window.dataLayer.push({
            event: name,
            vto_data: data
        });
    }
    console.log(`[VTO Analytics] ${name}`, data);
}
}

// Initializer
if (document.readyState === 'loading') {
document.addEventListener('DOMContentLoaded', () => window.NeonVTO = new NeonVTO());
} else {
window.NeonVTO = new NeonVTO();
}

import * as THREE from "three";
import {
FaceLandmarker,
FilesetResolver,
DrawingUtils,
type FaceLandmarkerResult
} from "@mediapipe/tasks-vision";

const video = document.querySelector<HTMLVideoElement>("#video")!;
const canvas = document.querySelector<HTMLCanvasElement>("#three")!;
const hud = document.querySelector<HTMLDivElement>("#hud")!;

// --- THREE setup (transparent canvas over video) ---
const renderer = new THREE.WebGLRenderer({ canvas, alpha: true, antialias: true });
renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));

const scene = new THREE.Scene();
const camera = new THREE.PerspectiveCamera(45, 1, 0.01, 100);
camera.position.set(0, 0, 1);

const light = new THREE.DirectionalLight(0xffffff, 1.0);
light.position.set(0.5, 1, 1);
scene.add(light);

const glasses = new THREE.Mesh(
new THREE.BoxGeometry(0.18, 0.06, 0.08),
new THREE.MeshStandardMaterial({ color: 0x00ffcc, metalness: 0.2, roughness: 0.4 })
);
scene.add(glasses);

// Maintain the "face anchor" as a group so you can add more accessories later.
const faceAnchor = new THREE.Group();
faceAnchor.add(glasses);
scene.add(faceAnchor);

// --- MediaPipe setup ---
async function initFaceLandmarker() {
// The official guide recommends FilesetResolver + FaceLandmarker.createFromOptions. :contentReference
const vision = await FilesetResolver.forVisionTasks(
"https://cdn.jsdelivr.net/npm/@mediapipe/tasks-vision@latest/wasm"
);

const faceLandmarker = await FaceLandmarker.createFromOptions(vision, {
baseOptions: {
// Use the official task model (host yourself for production).
modelAssetPath:
"https://storage.googleapis.com/mediapipe-models/face_landmarker/face_landmarker/float16/latest/face_landmarker.task"
},
outputFaceBlendshapes: false,
outputFacialTransformationMatrixes: true,
runningMode: "VIDEO",
numFaces: 1
});

return faceLandmarker;
}

async function startWebcam() {
const stream = await navigator.mediaDevices.getUserMedia({
video: { facingMode: "user", width: { ideal: 1280 }, height: { ideal: 720 } },
audio: false
});
video.srcObject = stream;
await video.play();
}

function resize() {
const w = canvas.clientWidth;
const h = canvas.clientHeight;
renderer.setSize(w, h, false);
camera.aspect = w / h;
camera.updateProjectionMatrix();
}
window.addEventListener("resize", resize);

function applyFaceMatrix(result: FaceLandmarkerResult) {
// faceTransformationMatrixes is provided when outputFacialTransformationMatrixes=true. :contentReference
const m = result.facialTransformationMatrixes?.[0]?.data;
if (!m) return false;

// MediaPipe returns a 4x4 matrix in a flat array (length 16). We feed it to Three.js.
// If orientation seems off, you may need to transpose depending on your camera pipeline.
const mat = new THREE.Matrix4().fromArray(Array.from(m));
faceAnchor.matrixAutoUpdate = false;
faceAnchor.matrix.copy(mat);

// Practical scaling/offset tuning:
// - scale down because face coords are typically in a metric-ish space relative to camera
// - move forward slightly (toward camera)
faceAnchor.scale.setScalar(0.35);
faceAnchor.position.z += 0.15;

return true;
}

async function main() {
hud.textContent = "Starting webcam…";
await startWebcam();
resize();

hud.textContent = "Loading face model…";
const faceLandmarker = await initFaceLandmarker();

const drawingUtils = new DrawingUtils(
// optional: if you want 2D debug overlay, provide a 2D canvas context
document.createElement("canvas").getContext("2d") as any
);

let lastTime = -1;

async function loop() {
requestAnimationFrame(loop);

if (video.readyState < 2) return;
const now = performance.now();
if (video.currentTime === lastTime) {
  renderer.render(scene, camera);
  return;
}
lastTime = video.currentTime;

const result = faceLandmarker.detectForVideo(video, now);
const ok = applyFaceMatrix(result);

hud.textContent = ok ? "Face tracked" : "No face";
renderer.render(scene, camera);
}

loop();
}

main().catch((e) => {
console.error(e);
hud.textContent = Error: ${String(e?.message ?? e)};
});
