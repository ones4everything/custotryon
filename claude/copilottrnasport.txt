  function normalizeHex(hex = '#ff00c8'){
    let h = hex.trim();
    if (h.startsWith('rgb')) return h;
    if (!h.startsWith('#')) h = `#${h}`;
    if (h.length === 4) h = `#${[...h.slice(1)].map(x => x + x).join('')}`;
    return h.slice(0,7);
  }
  function tint(hex, amt = 0.28){
    if (hex.startsWith('rgb')) return hex;
    const n = parseInt(hex.replace('#',''),16);
    let r = (n>>16)&255, g = (n>>8)&255, b = n&255;
    r = Math.min(255, Math.round(r + (255 - r) * amt));
    g = Math.min(255, Math.round(g + (255 - g) * amt));
    b = Math.min(255, Math.round(b + (255 - b) * amt));
    return `rgb(${r}, ${g}, ${b})`;
  }

    const base = normalizeHex(color || '#ff00c8');
    document.documentElement.style.setProperty('--current-glow', base);
    document.documentElement.style.setProperty('--glow', base);
    document.documentElement.style.setProperty('--glow2', tint(base));
  }

  function setState(header, open){
    const content = header.nextElementSibling;
    if(!content) return;
    
    header.classList.toggle('active', open);
    content.classList.toggle('active', open);
    
    if(open){
      // Get the full height of content

          content.style.maxHeight = content.scrollHeight + 'px';

        if(h !== header) setState(h, false);

      setState(header, !isActive);
      header.setAttribute('aria-expanded', !isActive ? 'true' : 'false');

      if (e.key === 'Enter' || e.key === ' ') {
